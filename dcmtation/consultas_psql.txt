-- Comandos PSQL para interactuar con los modelos del proyecto Freelance

-- 1. Listar todos los Clientes con su usuario asociado
SELECT c.id, c.name, c.email, u.username 
FROM freelance_core_client c
JOIN auth_user u ON c.user_id = u.id;

-- 2. Listar todos los Proyectos y el nombre de su Cliente
SELECT p.id, p.name AS proyecto, p.status, c.name AS cliente
FROM freelance_core_project p
JOIN freelance_core_client c ON p.client_id = c.id;

-- 3. Listar Tareas de un proyecto específico (por ejemplo, proyecto ID 1)
SELECT t.id, t.name, t.status, t.priority
FROM freelance_core_task t
WHERE t.project_id = 1;

-- 4. Contar cuántos proyectos hay en cada estado
SELECT status, COUNT(*) AS total
FROM freelance_core_project
GROUP BY status;

-- 5. Listar todos los Proyectos con su conteo de tareas
SELECT p.name AS proyecto, COUNT(t.id) AS total_tareas
FROM freelance_core_project p
LEFT JOIN freelance_core_task t ON p.id = t.project_id
GROUP BY p.id, p.name;

-- 6. Ver tareas pendientes ('todo') con su proyecto y cliente
SELECT t.name AS tarea, p.name AS proyecto, c.name AS cliente
FROM freelance_core_task t
JOIN freelance_core_project p ON t.project_id = p.id
JOIN freelance_core_client c ON p.client_id = c.id
WHERE t.status = 'todo';

-- 7. Listar Clientes y el número de proyectos que tienen
SELECT c.name, COUNT(p.id) AS num_proyectos
FROM freelance_core_client c
LEFT JOIN freelance_core_project p ON c.id = p.client_id
GROUP BY c.id, c.name;

-- 8. Ver detalle completo de una tarea y sus relaciones
SELECT 
    t.name AS tarea, 
    t.status AS estado_tarea, 
    p.name AS proyecto, 
    p.status AS estado_proyecto,
    c.name AS cliente
FROM freelance_core_task t
JOIN freelance_core_project p ON t.project_id = p.id
JOIN freelance_core_client c ON p.client_id = c.id;

-- Nota: Para ejecutar estos comandos, puedes entrar a la shell con:
-- ./db_shell.sh
-- Y luego pegar la consulta deseada.
